    printf("%d\n",i4);
    printf("-------------\n");
    // 4 int  和 long

    printf("%d\n",i9);
    printf("%d\n",sizeof(i9));

}

二、指针

指针是存储变量内存地址的变量

/* 重要：
     * 1  * 放在定义的变量前[int *v2]-->表示指向这个类型的指针--》定义指向这个类型的指针
     * 2  & 放在变量前--》表示 取这个变量的字---》&v1--》表示取v1的内存地址
     * 3  * 放在指针类型变量前，表示解引用
*/

1、定义指针类型变量和取变量地址

#include <stdio.h>

int main() {
    //  定义指针类型变量和取变量地址
    int v1=666;
    int *v2=&v1; // 定义了指针类型变量v2，指向了v1的地址，v2中存着v1的地址
    // v1和v2 不是一个东西，都不是一个类型，更不相等
    printf("v2的值是:%p",v2); // 0061FF18
    return 0;
}

6、指针类型参数

#include <stdio.h>


void changeA(int a){
    a=999;
    printf("内部是：%d\n",a);
}


void changePointA(int *a){
    *a=999;
    printf("内部是：%d\n",*a);

}
int main() {

    // 指针类型参数

    // 1  参数传递，修改值
    int a=1;
    // 调用changeA，修改a的值，a不会被真正的修改掉
    changeA(a);  // 把a复制了一份传进去了，里面改了，不会影响原来的值
    printf("外部是：%d\n",a);

    // 2  参数传递，指针类型修改值
    int a1=1;
    // 调用changePointA，修改a的值，a会被真正的修改掉
    changePointA(&a1);  // 把a1复制了一份,由于是个地址传进去了，在函数里面改了，改的是指向的位置，会影响原来的值
    printf("外部是：%d\n",a1);


    return 0;
}

三、结构体

# 一种用户自定义的数据类型，它允许存储不同类型的数据项
    -python中的列表，字典。。 本质都是c语言结构体指针

# 结构体中的数据成员可以是基本数据类型（如 int、float、char 等），也可以是其他结构体类型、指针类型等

1、基本使用

#include <stdio.h>


// 1 定义一个  人 结构体
struct Person{
    char name[30];
    int age;
};

int main() {
    //2  结构体定义和使用
    struct Person p1={"jsutin",88};
    printf("人名是:%s,年龄是：%d\n",p1.name,p1.age);

    struct Person p2={"彭于晏",34};
    printf("人名是:%s,年龄是：%d\n",p2.name,p2.age);


    // 3 结构体指针
    struct Person *p3=&p2;
    // 打印p3指向的地址人的名字
    printf("人名是：%s\n",(*p3).name); // 咱们理解的方式
    printf("人名是：%s\n",p3->name); //推荐使用这种
    return 0;
}

2、单向链表

#include <stdio.h>

// 定义一个Node结构体：要存数据，存下一个的地址
struct Node {
    int data;
    struct Node *next;
};

int main() {

    struct Node v3 = {33};
    struct Node v2 = {22, &v3};
    struct Node v1 = {11, &v2};

    printf("v1的数据为：%d\n", v1.data);                            // 11
    printf("v1的下一个元素数据为：%d\n", v1.next->data);              // 22
    printf("v1的下一个元素的下一个数据为：%d\n", v1.next->next->data); // 33

    return 0;
}

3、双向链表

#include <stdio.h>

// 定义一个Node结构体：要存数据，存下一个的地址
struct Node{
    int data;
    struct Node *next;
    struct Node *prev;
};

int main() {
    struct Node p3 = { 33 };
    struct Node p2 = { 22 };
    struct Node p1 = { 11 };

    // p1 的next 指向跑p2
    p1.next=&p2;

    // p2 的next 指向p3，  p2的prev指向p1
    p2.next=&p3;
    p2.prev=&p1;

    // p3 的next 是空， p3的prev指向p2
    p3.prev=&p2;


    // 随便用一个结构体，可以取出所有人
    printf("p1的值: %d\n", p1.data);
    printf("p2的值: %d\n", p1.next->data);
    printf("p3的值: %d\n", p1.next->next->data);

    printf("p3的值: %d\n", p3.data);
    printf("p2的值: %d\n", p3.prev->data);
    printf("p1的值: %d\n", p3.prev->prev->data);


    return 0;
}
